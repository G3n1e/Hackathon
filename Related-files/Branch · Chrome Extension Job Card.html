import express from "express";
import cors from "cors";

const app = express();
app.use(cors());
app.use(express.json({ limit: "2mb" }));

function makeJobCard({ title, url, text }) {
  // Placeholder “local model” logic:
  // For hackathon: simple heuristics + short summarization.
  // Replace later with a real local LLM call.
  const trimmed = (text || "").trim().slice(0, 4000);

  return {
    task_name: title || "Untitled Task",
    source_url: url,
    context: "Generated from a webpage for quick training reference.",
    when_to_use: "When performing this task or reviewing the procedure before starting.",
    tools_ppe: ["TBD (review site requirements)"],
    steps: [
      "Read the scope and confirm the work area is safe.",
      "Gather required tools/PPE and verify equipment condition.",
      "Follow the procedure summary and site-specific constraints.",
      "Perform a final check and document outcomes."
    ],
    common_mistakes: [
      "Skipping a safety check because the task seems routine",
      "Assuming the same steps apply across all sites without confirming"
    ],
    safety_notes: [
      "Stop work if conditions differ from expectations.",
      "Follow lockout/tagout and site safety rules where applicable."
    ],
    acceptance_checks: [
      "Work completed to spec",
      "Area left safe and clean",
      "Any deviations recorded"
    ],
    notes_from_source: trimmed ? trimmed.slice(0, 600) : "No text extracted.",
    youtube_link: "https://www.youtube.com/watch?v=VIDEO_ID_TBD"
  };
}

app.post("/jobcard", (req, res) => {
  try {
    const { title, url, text } = req.body || {};
    const job_card = makeJobCard({ title, url, text });
    res.json({ job_card });
  } catch (e) {
    res.status(500).json({ error: "Failed to generate job card." });
  }
});

app.listen(8787, "127.0.0.1", () => {
  console.log("Job Card service running on http://127.0.0.1:8787");
});

popup.js

JavaScript
async function getPageText() {
  const [tab] = await chrome.tabs.query({ active: true, currentWindow: true });

  const [{ result }] = await chrome.scripting.executeScript({
    target: { tabId: tab.id },
    func: () => {
      const selection = window.getSelection()?.toString()?.trim();
      if (selection && selection.length > 50) return selection;

      // Simple extraction: body innerText (fast + good enough for hackathon)
      const text = document.body?.innerText || "";
      return text.slice(0, 12000);
    }
  });

  return { tab, text: result || "" };
}

document.getElementById("generate").addEventListener("click", async () => {
  const out = document.getElementById("out");
  out.textContent = "Reading page…";

  const { tab, text } = await getPageText();
  out.textContent = "Sending to local model…";

  const resp = await fetch("http://127.0.0.1:8787/jobcard", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({
      title: tab.title,
      url: tab.url,
      text
    })
  });

  const data = await resp.json();
  const yt = document.getElementById("yt").value.trim();
  if (yt) data.job_card.youtube_link = yt;

  window._jobcard = data.job_card;
  out.textContent = JSON.stringify(data.job_card, null, 2);
});

document.getElementById("copy").addEventListener("click", async () => {
  if (!window._jobcard) return;
  await navigator.clipboard.writeText(JSON.stringify(window._jobcard, null, 2));
});
